    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
async function sha256(s){
  try {
    if (typeof crypto !== 'undefined' && crypto.subtle){
      const b = await crypto.subtle.digest('SHA-256', new TextEncoder().encode('khanhos:'+s));
      return [...new Uint8Array(b)].map(x=>x.toString(16).padStart(2,'0')).join('');
    }
  } catch {}
  let x = 5381; for (let i = 0; i < s.length; i++) x = (Math.imul(x, 33) ^ s.charCodeAt(i)) >>> 0;
  return 'fb' + x.toString(16).padStart(8,'0') + s.length.toString(16);
}

/* ---------- 2. MARKDOWN RENDERER (escape an toàn) ---------- */
function inlineMD(t){
  t = esc(t);
  t = t.replace(/`([^`]+)`/g, (m,c)=>`<code>${c}</code>`);
  t = t.replace(/\*\*([^*]+)\*\*/g,'<strong>$1</strong>');
  t = t.replace(/~~([^~]+)~~/g,'<del>$1</del>');
  t = t.replace(/\*([^*]+)\*/g,'<em>$1</em>');
  t = t.replace(/\[([^\]]+)\]\((https?:[^)\s]+)\)/g,'<a href="$2" target="_blank" rel="noopener">$1</a>');
  return t;
}
function codeBlockHTML(code, lang){
  return `<div class="codeblock"><div class="cb-head"><span class="cb-lang">${esc(lang||'text')}</span>`+
    `<button class="cb-copy">${icon('copy',13)}<span>SAO CHÉP</span></button></div>`+
    `<pre><code>${highlight(code, lang)}</code></pre></div>`;
}
function renderMD(src){
  const lines = src.split('\n');
  let html = '', i = 0, inCode = false, codeLang = '', codeBuf = [];
  const blank = l => !l.trim();
  while (i < lines.length){
    const line = lines[i];
    const isFence = /^```\s*\w*\s*$/.test(line.trim());
    if (isFence){
      if (!inCode){ inCode = true; codeLang = line.trim().replace(/^```\s*/,'').toLowerCase(); codeBuf = []; i++; continue; }
      html += codeBlockHTML(codeBuf.join('\n'), codeLang);
      inCode = false; i++; continue;
    }
    if (inCode){ codeBuf.push(line); i++; continue; }
    if (blank(line)){ i++; continue; }
    let m;
    if ((m = line.match(/^(#{1,4})\s+(.*)/))){
      const lv = m[1].length;
      html += `<h${lv}>${inlineMD(m[2])}</h${lv}>`; i++; continue;
    }
    if (/^(-{3,}|\*{3,})\s*$/.test(line.trim())){ html += '<hr>'; i++; continue; }
    if (/^\|/.test(line.trim())){
      const rows = [];
      while (i < lines.length && /^\|/.test(lines[i].trim())){ rows.push(lines[i].trim()); i++; }
      const cells = r => r.replace(/^\||\|$/g,'').split('|').map(c=>c.trim());
      let thead = '', tbody = '';
      rows.forEach(r => {
        const cs = cells(r);
        const isSep = cs.every(c => /^:?-{2,}:?$/.test(c));
        if (isSep) return;
        if (!thead){ thead = '<tr>'+cs.map(c=>`<th>${inlineMD(c)}</th>`).join('')+'</tr>'; }
        else tbody += '<tr>'+cs.map(c=>`<td>${inlineMD(c)}</td>`).join('')+'</tr>';
      });
      html += `<div class="table-wrap"><table>${thead}${tbody?`<tbody>${tbody}</tbody>`:''}</table></div>`;
      continue;
    }
    if (/^[-*]\s+/.test(line)){
      const items = [];
      while (i < lines.length && /^[-*]\s+/.test(lines[i])){ items.push(lines[i].replace(/^[-*]\s+/,'')); i++; }
      html += '<ul>'+items.map(t=>`<li>${inlineMD(t)}</li>`).join('')+'</ul>'; continue;
    }
    if (/^\d+\.\s+/.test(line)){
      const items = [];
      while (i < lines.length && /^\d+\.\s+/.test(lines[i])){ items.push(lines[i].replace(/^\d+\.\s+/,'')); i++; }
      html += '<ol>'+items.map(t=>`<li>${inlineMD(t)}</li>`).join('')+'</ol>'; continue;
    }
    if (/^>\s?/.test(line)){
      const buf = [];
      while (i < lines.length && /^>\s?/.test(lines[i])){ buf.push(lines[i].replace(/^>\s?/,'')); i++; }
      html += `<blockquote><p>${inlineMD(buf.join(' '))}</p></blockquote>`; continue;
    }
    const buf = [];
    while (i < lines.length && !blank(lines[i]) && !/^([#>*-]|\d+\.\s|\||```)/.test(lines[i].trim())){ buf.push(lines[i]); i++; }
    if (buf.length){ html += `<p>${inlineMD(buf.join(' '))}</p>`; }
    else i++;
  }
  if (inCode) html += codeBlockHTML(codeBuf.join('\n'), codeLang);
  return html;
}
const CARET = '<span class="caret" aria-hidden="true"></span>';
function withCaret(html){
  if (!html) return `<p>${CARET}</p>`;
  const m = html.match(/<\/(p|li|h[1-4]|pre|blockquote)>$/);
  if (m){
    const tag = m[1], at = html.lastIndexOf('</'+tag+'>');
    return html.slice(0, at) + CARET + html.slice(at);
  }
  return html + CARET;
}

const RE_STR  = /'(?:[^'\\\n]|\\.)*'|"(?:[^"\\\n]|\\.)*"|`(?:[^`\\]|\\.)*`/.source;
const RE_NUM  = /\b\d+(?:\.\d+)?\b/.source;
const RE_FN   = /\b[A-Za-z_]\w*(?=\s*\()/.source;
const RE_COMJ = /\/\/[^\n]*|\/\*[\s\S]*?\*\//.source;
const RE_COMH = /(?:#|\/\/)[^\n]*/.source;
const KW = {
  js: 'const|let|var|function|return|if|else|for|while|do|switch|case|break|continue|new|class|extends|import|from|export|default|async|await|try|catch|finally|throw|typeof|instanceof|interface|type|enum|implements|public|private|readonly|static|of|in|yield|delete|true|false|null|undefined|this|super',
  py: 'def|return|if|elif|else|for|while|class|import|from|as|try|except|finally|raise|with|lambda|pass|yield|async|await|global|not|and|or|is|in|del|assert|True|False|None',
  sql: 'SELECT|FROM|WHERE|JOIN|LEFT|RIGHT|INNER|OUTER|ON|GROUP|BY|ORDER|LIMIT|OFFSET|INSERT|INTO|VALUES|UPDATE|SET|DELETE|CREATE|TABLE|PRIMARY|KEY|FOREIGN|REFERENCES|NOT|NULL|DEFAULT|UNIQUE|INDEX|AND|OR|AS',
  bash: 'if|then|fi|else|for|do|done|while|case|esac|function|export|return|local|echo|cd|npm|npx|pnpm|git|node',
  prisma: 'model|generator|datasource|provider|enum|String|Int|Float|Boolean|DateTime|Json',
};
const FAMILY = { js:'js', jsx:'js', ts:'js', tsx:'js', typescript:'js', javascript:'js', py:'py', python:'py', sql:'sql', bash:'bash', sh:'bash', shell:'bash', prisma:'prisma' };
function highlight(code, lang){
  const fam = FAMILY[String(lang||'').toLowerCase()];
  if (!fam || fam === 'plain') return esc(code);
  const com = (fam === 'py' || fam === 'bash' || fam === 'prisma') ? RE_COMH : RE_COMJ;
  const kw = KW[fam] || '';
  const kwAlt = kw ? `\\b(?:${kw})\\b` : '\\b\\B';
  const master = new RegExp(`(?<com>${com})|(?<str>${RE_STR})|(?<num>${RE_NUM})|(?<kw>${kwAlt})|(?<fn>${RE_FN})`, fam === 'sql' ? 'gi' : 'g');
  let out = '', last = 0;
  for (const m of code.matchAll(master)){
    out += esc(code.slice(last, m.index));
    const g = m.groups;
    const cls = g.com ? 'c-com' : g.str ? 'c-str' : g.num ? 'c-num' : g.kw ? 'c-kw' : 'c-fn';
    out += `<span class="${cls}">${esc(m[0])}</span>`;
    last = m.index + m[0].length;
  }
  return out + esc(code.slice(last));
}

function extractCode(prompt){
  const m = prompt.match(/```[a-z]*\n([\s\S]*?)```/i);
  if (m) return m[1];
  if (/[;{}]/.test(prompt) && /\b(function|const|let|class|def|import|export|SELECT|INSERT|module)\b/i.test(prompt)) return prompt;
  return null;
}
function reviewCode(code){
  const out = [];
  for (const rule of REVIEW_RULES){
    const re = new RegExp(rule.re.source, 'g' + (rule.re.flags.includes('i') ? 'i' : ''));
    const m = re.exec(code);
    if (m){
      const line = code.slice(0, m.index).split('\n').length;
      out.push({ sev: rule.sev, line, msg: rule.msg, fix: rule.fix });
    }
  }
  return out;
}

function buildVideoParams(prompt, seed){
  const rng = mulberry32((hashStr(prompt) ^ Math.imul(seed, 2654435761)) >>> 0);

function drawVideoFrame(ctx, w, h, P, t){
  const g = ctx.createLinearGradient(0, 0, w, h);
  g.addColorStop(0, P.pal.bg[0]); g.addColorStop(1, P.pal.bg[1]);
  ctx.fillStyle = g; ctx.fillRect(0, 0, w, h);
  P.arcs.forEach(a => {
    const ang = a.ph + t*Math.PI*2*a.sp;
    ctx.beginPath();
    ctx.arc(P.cx, P.cy, a.r, ang, ang + 0.9 + 2.2*Math.min(1, a.w/40));
    ctx.strokeStyle = a.col; ctx.globalAlpha = a.a; ctx.lineWidth = a.w; ctx.stroke();
  });
  P.dots.forEach(d => {
    const ang = d.ph + t*Math.PI*2*d.sp;
    const x = P.cx + Math.cos(ang)*d.r, y = P.cy + Math.sin(ang)*d.r*0.62;
    ctx.globalAlpha = d.a; ctx.fillStyle = d.col;
    ctx.beginPath(); ctx.arc(x, y, d.size, 0, 6.28); ctx.fill();
  });
  ctx.globalAlpha = 0.5; ctx.fillStyle = P.pal.light;
  ctx.fillRect(0, h*0.82, w, 1);
  ctx.globalAlpha = 1;
}
async function recordVideo(prompt, seed, onPct, signal){
  const cv = document.createElement('canvas'); cv.width = 640; cv.height = 360;
  const ctx = cv.getContext('2d');
  const P = buildVideoParams(prompt, seed);
  drawVideoFrame(ctx, 640, 360, P, 0);
  const supported = 'MediaRecorder' in window && typeof cv.captureStream === 'function';
  let mime = '';
  if (supported){
    mime = ['video/webm;codecs=vp9','video/webm;codecs=vp8','video/webm','video/mp4']
      .find(m => { try { return MediaRecorder.isTypeSupported(m); } catch { return false; } }) || '';
  }
  if (!supported || !mime) return { previewOnly: true };
  const stream = cv.captureStream(30);
  const rec = new MediaRecorder(stream, mime ? { mimeType: mime } : undefined);
  const chunks = [];
  rec.ondataavailable = e => { if (e.data && e.data.size) chunks.push(e.data); };
  const stopped = new Promise(res => { rec.onstop = res; });
  rec.start(250);
  const DUR = 3200, t0 = performance.now();
  await new Promise(resolve => {
    function frame(now){
      const el = now - t0, t = Math.min(1, el/DUR);
      drawVideoFrame(ctx, 640, 360, P, t);
      onPct(Math.min(88, Math.round(t*88)));
      if (signal.aborted || el >= DUR){ resolve(); return; }
      requestAnimationFrame(frame);
    }
    requestAnimationFrame(frame);
  });
  try { rec.stop(); } catch {}
  await stopped;
  if (signal.aborted) throw new Error('abort');
  const blob = new Blob(chunks, { type: mime });
  if (!blob.size) throw { provider: 503 };
  return { url: URL.createObjectURL(blob), size: blob.size, mime };
}

function parseDur(str){
  const s = norm(str);
  const m = s.match(/^(\d+)\s*(gio|hour|h|thang|month|tuan|week|ngay|day|d|t|w|g)$/);
  if (!m) return null;
  const n = parseInt(m[1], 10);
  if (n <= 0 || n > 999) return null;
  const u = m[2];
  let ms, label;
  if (u === 'h' || u === 'gio' || u === 'g'){ ms = n * 3600e3; label = n + ' giờ'; }
  else if (u === 'd' || u === 'day' || u === 'ngay'){ ms = n * 86400e3; label = n + ' ngày'; }
  else if (u === 'w' || u === 'tuan'){ ms = n * 604800e3; label = n + ' tuần'; }
  else { ms = n * 2592000e3; label = n + ' tháng'; }
  return { ms, label };
}

    pushSys(text, [{ t:'err', s:'lệnh không nhận diện được — gõ /help để xem danh sách lệnh' }]);
    return;
  }
  const a = acct();
  if (!isOwner(a)){
    pushSys(text, [{ t:'err', s:'lệnh quản trị chỉ dành cho owner' }]);
    return;
  }
  if (cmd === 'give'){
    const planKey = planAlias(parts[1] || '');
    const email = (parts[2] || '').toLowerCase();
    const durTok = parts.slice(3).join(' ');
    if (!planKey || !email || !durTok){
      pushSys(text, [
        { t:'err', s:'thiếu tham số — dùng: /give plus|vip|vipplus|max <email> <thời gian>' },
        { t:'dim', s:'ví dụ: /give vip ban@example.com 1thang · /give plus ban@example.com 5h' },
      ]);
      return;
    }
    const dur = parseDur(durTok);
    if (!dur){
      pushSys(text, [
        { t:'err', s:'thời gian không hợp lệ: "' + durTok + '"' },
        { t:'dim', s:'định dạng: 5h · 1day · 3day · 1thang · 2thang (số + đơn vị)' },
      ]);
      return;
    }
    const target = findUserByEmail(email);
    if (!target){
      pushSys(text, [{ t:'err', s:'email chưa đăng ký: ' + email }]);
      return;
    }
    target.plan = planKey;
    target.planExpires = Date.now() + dur.ms;
    persist(); renderSidebar();
    pushSys(text, [
      { t:'ok', s: PLANS[planKey].name + ' → ' + target.email },
      { t:'dim', s:'hết hạn: ' + fmtExpiry(target.planExpires) + ' (' + dur.label + ')' },
    ]);
    return;
  }
  // /ungoi <email>
  const email = (parts[1] || '').toLowerCase();
  if (!email){
    pushSys(text, [{ t:'err', s:'thiếu email — dùng: /ungoi <email>' }]);
    return;
  }
  const target = findUserByEmail(email);
  if (!target){
    pushSys(text, [{ t:'err', s:'email chưa đăng ký: ' + email }]);
    return;
  }
  target.plan = 'free';
  target.planExpires = null;
  persist(); renderSidebar();
  pushSys(text, [{ t:'ok', s:'đã thu hồi gói của ' + target.email + ' — về FREE' }]);
}

async function streamTokens(text, asMsg, ctrl, prefix){
  const tokens = text.match(/\s+|\S+/g) || [];
  let buf = prefix || '';
  asMsg.status = 'streaming';
  for (const tk of tokens){
    if (ctrl.signal.aborted) break;
    buf += tk; asMsg.content = buf;
    refreshMsg(asMsg); scrollBottom(false);
    let d = 10 + Math.random() * 16;
    if (/^\n\n+$/.test(tk)) d += 110;
    else if (/[!?…]$/.test(tk)) d += 90;
    await wait(d * state.settings.speed, ctrl.signal);
  }
}

async function startTurn(userContent){
  if (gen.running) return;
  const mode = state.mode;
  if (MODES[mode].kind === 'text' && tokensLeft() < TEXT_MIN_RESERVE){
    toast('Hết hạn mức tháng của gói ' + curPlan().name + ' — mua thêm ở trang Tài khoản.', 'err');
    return;
  }
  const a = acct();
  let chat = getActiveChat();
  if (!chat){ chat = { id: uid(), title: null, createdAt: Date.now(), messages: [] }; a.chats.unshift(chat); a.activeChatId = chat.id; }
  chat.messages.push({ id: uid(), role: 'user', content: userContent, mode, createdAt: Date.now() });
  if (!chat.title) chat.title = extractSubject(userContent);
  const kind = MODES[mode].kind;
  const asMsg = { id: uid(), role: 'assistant', kind, mode, content: '', status: 'routing', model: null, route: null, thinking: null, pipeline: null, createdAt: Date.now(), errorCount: 0 };
  if (kind === 'image') asMsg.image = { prompt: userContent, seed: 100000 + rnd(900000), hadRef: !!refImage, _ref: refImage ? refImage.img : null };
  if (kind === 'video') asMsg.video = { prompt: userContent, seed: 100000 + rnd(900000) };
  chat.messages.push(asMsg);
  clearRef();
  persist(); renderSidebar(); renderChat();
  if (kind === 'text') await driveAssistant(chat, asMsg);
  else await driveGeneration(chat, asMsg);
}
async function driveAssistant(chat, asMsg){
  const ctrl = new AbortController();
  gen = { ctrl, running: true };
  setStreamingUI(true, (asMsg.model || '') + ' đang trả lời…');
  const idx0 = chat.messages.findIndex(x => x.id === asMsg.id);
  let prompt = '';
  for (let i = idx0 - 1; i >= 0; i--) if (chat.messages[i].role === 'user'){ prompt = chat.messages[i].content; break; }
  const t0 = performance.now();
  try {
    const rl = checkRate(asMsg.mode);
    if (!rl.ok){
      asMsg.status = 'error';
      asMsg.error = { code: 429, msg: 'Rate limit — mode ' + asMsg.mode + ' quá nhiều request trong 10 giây. Chờ ~' + rl.retryAfter + 's rồi thử lại.' };
      refreshMsg(asMsg); return;
    }
    await wait(200 + rnd(300), ctrl.signal);
    if (ctrl.signal.aborted) throw new Error('abort');
    if (asMsg.errorCount === 0 && Math.random() < 0.05){
      asMsg.status = 'error';
      asMsg.error = { code: 502, msg: 'Router không phản hồi sau 5000ms. Kiểm tra biến môi trường provider, mạng, rồi thử lại.' };
      refreshMsg(asMsg); return;
    }
    const { model, pinned, provider } = resolveTextModel(asMsg.mode);
    const task = routeMessage(prompt);
    const label = { chat: task.label, reasoner: 'deep-reasoning', coder: 'coding-pipeline', reviewer: 'code-review' }[asMsg.mode];
    asMsg.route = { label, model, pinned, provider, score: asMsg.mode === 'chat' ? task.score : null, ms: task.ms };
    asMsg.model = model;
    refreshMsg(asMsg);
    if (asMsg.mode === 'coder'){
      await drivePipeline(chat, asMsg, prompt, ctrl);
    } else {
      const wantThink = asMsg.mode === 'reasoner' || (asMsg.mode === 'chat' && (model === 'glm-5.3' || pinned));
      if (wantThink && state.settings.showThinking){
        asMsg.status = 'thinking'; asMsg.thinking = []; refreshMsg(asMsg);
        const th0 = performance.now();
        for (const step of (thinkingFor(asMsg.mode === 'reasoner' ? 'reasoning' : task.task) || [])){
          await wait(340 + rnd(380), ctrl.signal);
          if (ctrl.signal.aborted) throw new Error('abort');
          asMsg.thinking.push(step); refreshMsg(asMsg); scrollBottom(false);
        }
        asMsg.thinkDur = fmtDur(performance.now() - th0);
      }
      const reply = asMsg.mode === 'reviewer' ? buildReviewReport(prompt) : buildReply(prompt, task);
      await streamTokens(reply.body, asMsg, ctrl);
    }
    if (ctrl.signal.aborted) throw new Error('abort');
    asMsg.status = 'done';
    asMsg.meta = { tok: Math.round((asMsg.content || '').split(/\s+/).filter(Boolean).length * 1.35), dur: fmtDur(performance.now() - t0) };
  } catch (e) {
    if (asMsg.pipeline) asMsg.pipeline.forEach(s => { if (s.status === 'running') s.status = 'stopped'; });
  } finally {
    gen = { ctrl: null, running: false };
    setStreamingUI(false);
    if (asMsg.status !== 'error' && asMsg.status !== 'done') asMsg.status = 'stopped';
    if (!asMsg.content && !asMsg.pipeline && !asMsg.thinking && asMsg.status === 'stopped'){
      const i = chat.messages.findIndex(x => x.id === asMsg.id);
      if (i > -1) chat.messages.splice(i, 1);
    }
    if (asMsg.status === 'done'){
      chargeTokens(Math.max(1, Math.round((asMsg.content || '').split(/\s+/).filter(Boolean).length * 1.35)));
    }
    persist(); renderSidebar();
    if (getActiveChat()) renderChat();
    aiLog({
      requestId: uid(), userId: acct().email, mode: asMsg.mode,
      model: asMsg.model, provider: asMsg.route && asMsg.route.provider,
      durationMs: Math.round(performance.now() - t0), status: asMsg.status,
      usage: { tokens: (asMsg.meta && asMsg.meta.tok) || 0 },
    });
  }
}
async function drivePipeline(chat, asMsg, prompt, ctrl){
  asMsg.pipeline = [
    { k: 'reasoner', name: 'REASONER',    status: 'pending' },
    { k: 'coder',    name: 'CODER',       status: 'pending' },
    { k: 'reviewer', name: 'REVIEWER',    status: 'pending' },
    { k: 'final',    name: 'CODER FINAL', status: 'pending' },
  ];
  const st = k => asMsg.pipeline.find(s => s.k === k);
  const s1 = st('reasoner'); s1.t0 = performance.now(); s1.steps = []; s1.status = 'running'; refreshMsg(asMsg);
  const plan = [...(thinkingFor('code') || []),
    'chốt cấu trúc output: lý do ngắn → code → lưu ý bảo mật',
    'giao reviewer checklist: auth, biên lỗi, hiệu năng'];
  for (const step of plan){
    await wait(300 + rnd(280), ctrl.signal);
    if (ctrl.signal.aborted) throw new Error('abort');
    s1.steps.push(step); refreshMsg(asMsg); scrollBottom(false);
  }
  s1.ms = performance.now() - s1.t0; s1.status = 'done'; refreshMsg(asMsg);
  const s2 = st('coder'); s2.t0 = performance.now(); s2.status = 'running'; refreshMsg(asMsg);
  const tpl = buildReply(prompt, { task: 'code' });
  await streamTokens(tpl.body, asMsg, ctrl);
  if (ctrl.signal.aborted) throw new Error('abort');
  s2.ms = performance.now() - s2.t0; s2.status = 'done'; refreshMsg(asMsg);
  const s3 = st('reviewer'); s3.t0 = performance.now(); s3.findings = []; s3.status = 'running'; refreshMsg(asMsg);
  const pool = PIPE_FINDINGS[tpl.key] || PIPE_FINDINGS.default;
  const willFlag = Math.random() < 0.7;
  const findings = willFlag

  const hasReal = findings.some(f => ['CRITICAL','HIGH','MEDIUM'].includes(f.sev));

  } else {

    dur: fmtDur((s1.ms || 0) + (s2.ms || 0) + (s3.ms || 0) + (s4.ms || 0)),
  };

async function driveGeneration(chat, asMsg){
  const ctrl = new AbortController();
  gen = { ctrl, running: true };
  const kind = asMsg.kind, mode = asMsg.mode;
  const info = asMsg.image || asMsg.video;
  setStreamingUI(true, MODES[mode].label.toLowerCase() + ' · job ' + (info ? info.seed : ''));
  const t0 = performance.now();
  try {
    const rl = checkRate(mode);
    if (!rl.ok){
      asMsg.status = 'error';
      asMsg.error = { code: 429, msg: 'Rate limit — mode ' + mode + ' quá nhiều request trong 10 giây. Chờ ~' + rl.retryAfter + 's rồi thử lại.' };
      refreshMsg(asMsg); return;
    }
    const cost = kind === 'image' ? IMAGE_TOKEN_COST : VIDEO_TOKEN_COST;
    if (tokensLeft() < cost){
      asMsg.status = 'error';
      asMsg.error = { code: 402, msg: 'Hạn mức tháng của gói ' + curPlan().name + ' không đủ để tạo ' + (kind === 'image' ? 'ảnh' : 'video') + ' — mua thêm ở trang Tài khoản.' };
      refreshMsg(asMsg); return;
    }
    if (Math.random() < 0.05){
      asMsg.status = 'error';
      asMsg.error = { code: 503, msg: 'Provider ' + AI_ENV[kind.toUpperCase() + '_PROVIDER'] + ' trả 503 — job không khởi tạo được. Bấm Thử lại.' };
      refreshMsg(asMsg); return;
    }
    asMsg.status = 'generating'; asMsg.phase = 'PREPARING'; asMsg.pct = 0;
    asMsg.model = AI_ENV[kind.toUpperCase() + '_MODEL'];
    asMsg.route = { label: mode, model: asMsg.model, provider: AI_ENV[kind.toUpperCase() + '_PROVIDER'], seed: info.seed, ms: 0 };
    refreshMsg(asMsg); scrollBottom(true);
    await wait(420 + rnd(300), ctrl.signal);
    if (ctrl.signal.aborted) throw new Error('abort');
    asMsg.phase = 'GENERATING'; asMsg.pct = 6; refreshMsg(asMsg);
    if (kind === 'image'){
      for (let s = 1; s <= 4; s++){
        await wait(460 + rnd(240), ctrl.signal);
        if (ctrl.signal.aborted) throw new Error('abort');
        asMsg.pct = 6 + Math.round(s * 21);
        refreshMsg(asMsg); scrollBottom(false);
      }
    } else {
      let lastPaint = 0;
      const res = await recordVideo(info.prompt, info.seed, p => {
        asMsg.pct = p;
        const now = performance.now();
        if (now - lastPaint > 140){ lastPaint = now; refreshMsg(asMsg); scrollBottom(false); }
      }, ctrl.signal);
      if (res.previewOnly) asMsg.video.previewOnly = true;
      else { asMsg.video.url = res.url; asMsg.video.size = res.size; asMsg.video.mime = res.mime; }
    }
    if (ctrl.signal.aborted) throw new Error('abort');
    asMsg.phase = 'PROCESSING'; asMsg.pct = 93; refreshMsg(asMsg);
    await wait(380 + rnd(220), ctrl.signal);
    if (ctrl.signal.aborted) throw new Error('abort');
    asMsg.status = 'done'; asMsg.phase = 'COMPLETED'; asMsg.pct = 100;
    asMsg.route.ms = Math.round(performance.now() - t0);
    refreshMsg(asMsg);
  } catch (e) {
    if (e && e.provider){
      asMsg.status = 'error';
      asMsg.error = { code: e.provider, msg: 'Provider lỗi khi encode video (blob rỗng) — bấm Thử lại để chạy job mới.' };
    }
  } finally {
    gen = { ctrl: null, running: false };
    setStreamingUI(false);
    if (asMsg.status !== 'done' && asMsg.status !== 'error') asMsg.status = 'stopped';
    if (asMsg.status === 'done'){
      chargeTokens(kind === 'image' ? IMAGE_TOKEN_COST : VIDEO_TOKEN_COST);
    }
    persist(); renderSidebar();
    if (getActiveChat()) renderChat();
    aiLog({
      requestId: uid(), userId: acct().email, mode, model: asMsg.model,
      provider: asMsg.route && asMsg.route.provider,
      durationMs: Math.round(performance.now() - t0), status: asMsg.status,
      usage: { generations: asMsg.status === 'done' ? 1 : 0, seed: info.seed },
    });
  }
}

function extractSubject(p){
  let s = p.trim().replace(/[?！!]+$/,'');
  for (let i = 0; i < 2; i++){
    s = s.replace(/^\s*(?:xin\s+)?(?:viet|viết|lam|làm|giup|giúp|cho|check|lam|hỏi|hoi)?\s*(?:cho|giup|giúp)?\s*(?:tao|toi|tôi|tui|minh|mình|to|anh|em|chi|chị)?\s*/i, '');
  }
  s = s.trim();
  if (!s) return 'Cuộc trò chuyện mới';
  if (s.length > 42){
    const cut = s.slice(0, 42), at = cut.lastIndexOf(' ');
    s = (at > 18 ? cut.slice(0, at) : cut) + '…';
  }
  return s.charAt(0).toUpperCase() + s.slice(1);
}

const modelDrop = $('#modelDrop');
function renderModeDrop(){
  const kind = MODES[state.mode].kind;
  const tick = `<span class="mo-tick">${icon('check',14)}</span>`;
  const modeHTML = Object.entries(MODES).map(([id, m]) => `
    <div class="mopt ${state.mode === id ? 'sel' : ''}" data-mode="${id}">
      <span class="mo-ic">${icon(m.icon, 15)}</span>
      <span class="mo-name">${m.label}</span>
      ${state.mode === id ? tick : ''}
      <span class="mo-desc">${m.desc}</span>
    </div>`).join('');
  const lock = kind !== 'text';
  const modelHTML = MODELS.map(o => `
    <div class="mopt ${state.modelChoice === o.id ? 'sel' : ''} ${lock ? 'dis' : ''}" data-model="${o.id}">
      <span class="mo-name">${o.name}</span>
      ${state.modelChoice === o.id ? tick : ''}
      <span class="mo-desc">${o.desc}</span>
    </div>`).join('');
  modelDrop.innerHTML = `
    <div class="md-group">AI MODE</div>${modeHTML}
    <div class="md-group">MODEL ${lock ? '· SERVER GIỮ CHO IMAGE/VIDEO' : '· CHỈ ÁP DỤNG MODE TEXT'}</div>${modelHTML}`;
}
function renderModeUI(){
  const m = MODES[state.mode];
  const kind = m.kind;
  const modelTxt = kind === 'text'
    ? (state.modelChoice !== 'auto' ? state.modelChoice : resolveTextModel(state.mode).model)
    : AI_ENV[kind.toUpperCase() + '_MODEL'];
  $('#modelLabel').textContent = m.label.toLowerCase() + ' · ' + modelTxt;
  ta.placeholder = PLACEHOLDERS[state.mode];
  $('#hintText').textContent = MODE_HINTS[state.mode];
  uploadBtn.style.display = state.mode === 'image' ? 'grid' : 'none';
  if (state.mode !== 'image') clearRef();
}

stage.addEventListener('click', async e => {
  const sug = e.target.closest('[data-sug]');
  if (sug && !gen.running){
    const md = sug.dataset.mode;
    if (md && MODES[md] && md !== state.mode){ state.mode = md; persist(); renderModeUI(); }
    ta.value = sug.dataset.sug; autoGrow(); send();
    return;
  }
  const cb = e.target.closest('.cb-copy');
  if (cb){
    const code = cb.closest('.codeblock').querySelector('pre code').textContent;
    await copyText(code);
    cb.classList.add('done');
    cb.innerHTML = icon('check',13) + '<span>ĐÃ CHÉP</span>';
    setTimeout(() => { cb.classList.remove('done'); cb.innerHTML = icon('copy',13) + '<span>SAO CHÉP</span>'; }, 1400);
    return;
  }
  const act = e.target.closest('[data-act]');
  if (!act) return;
  const a = act.dataset.act, id = act.dataset.id;
  const aCC = acct();
  const chat = aCC.chats.find(c => c.id === aCC.activeChatId); if (!chat) return;
  const msg = chat.messages.find(m => m.id === id); if (!msg) return;

  if (a === 'copy'){
    await copyText(msg.content || (msg.image && msg.image.prompt) || (msg.video && msg.video.prompt) || '');
    toast('Đã sao chép');
  }
  if (a === 'download'){
    if (msg.kind === 'image'){
      const cv = document.createElement('canvas'); cv.width = 640; cv.height = 640;
      drawGenerative(cv, msg.image.prompt, msg.image.seed, null);
      const link = document.createElement('a');
      link.download = 'khanhos-image-' + msg.image.seed + '.png';
      link.href = cv.toDataURL('image/png');
      link.click();
      toast('Đã tải ảnh PNG 640×640');
    } else if (msg.kind === 'video' && msg.video && msg.video.url){
      const ext = (msg.video.mime || '').includes('mp4') ? 'mp4' : 'webm';
      const link = document.createElement('a');
      link.download = 'khanhos-video-' + msg.video.seed + '.' + ext;
      link.href = msg.video.url;
      link.click();
      toast('Đã tải video ' + ext.toUpperCase());
    }
  }
  if (a === 'regen' && !gen.running){
    const i = chat.messages.findIndex(m => m.id === id);
    const target = chat.messages[i];
    chat.messages.splice(i);
    let prompt = '';
    for (let j = chat.messages.length - 1; j >= 0; j--) if (chat.messages[j].role === 'user'){ prompt = chat.messages[j].content; break; }
    const mode = target.mode || state.mode;
    const kind = target.kind || 'text';
    const fresh = { id: uid(), role: 'assistant', kind, mode, content: '', status: 'routing', model: null, route: null, thinking: null, pipeline: null, createdAt: Date.now(), errorCount: 0 };
    if (kind === 'image') fresh.image = { prompt, seed: 100000 + rnd(900000) };
    if (kind === 'video') fresh.video = { prompt, seed: 100000 + rnd(900000) };
    chat.messages.push(fresh);
    persist(); renderChat();
    if (kind === 'text') driveAssistant(chat, fresh); else driveGeneration(chat, fresh);
  }
  if (a === 'retry' && !gen.running){
    msg.errorCount = 1; msg.status = 'routing'; msg.error = null; msg.phase = null; msg.pct = 0;
    refreshMsg(msg);
    if (msg.kind === 'image' || msg.kind === 'video') driveGeneration(chat, msg);
    else driveAssistant(chat, msg);
  }
  if (a === 'edit' && !gen.running){
    const el = stage.querySelector(`[data-mid="${id}"]`);
    if (!el) return;
    const body = el.querySelector('.user-body');
    if (!body) return;
    body.outerHTML = `
      <div class="editbox">
        <textarea id="editTa">${esc(msg.content)}</textarea>
        <div class="edit-acts">
          <button class="btn ghost" data-act="edit-cancel">Huỷ</button>
          <button class="btn acc" data-act="edit-save" data-id="${id}">Lưu &amp; gửi lại</button>
        </div>
      </div>`;
    const eta = $('#editTa');
    eta.focus(); eta.setSelectionRange(eta.value.length, eta.value.length);
    const eb = el.querySelector('[data-act="edit"]');
    if (eb) eb.style.display = 'none';
  }
  if (a === 'edit-cancel'){ renderChat(); }
  if (a === 'edit-save' && !gen.running){
    const v = ($('#editTa') || {}).value || '';
    if (!v.trim()){ renderChat(); return; }
    msg.content = v.trim();
    const i = chat.messages.findIndex(m => m.id === id);
    const cut = chat.messages.slice(i + 1);
    const prevAs = [...cut].reverse().find(m => m.role === 'assistant');
    const kind = (prevAs && prevAs.kind) || 'text';
    const mode = (prevAs && prevAs.mode) || msg.mode || state.mode;
    chat.messages.splice(i + 1);
    const fresh = { id: uid(), role: 'assistant', kind, mode, content: '', status: 'routing', model: null, route: null, thinking: null, pipeline: null, createdAt: Date.now(), errorCount: 0 };
    if (kind === 'image') fresh.image = { prompt: v.trim(), seed: 100000 + rnd(900000) };
    if (kind === 'video') fresh.video = { prompt: v.trim(), seed: 100000 + rnd(900000) };
    chat.messages.push(fresh);
    persist(); renderChat();
    if (kind === 'text') driveAssistant(chat, fresh); else driveGeneration(chat, fresh);
  }
});
